```sql
WITH s AS (
  SELECT vvh_id, beginn, bundesland, oenace, lag("oenace") OVER w AS "oenace_vor", lag("bundesland") OVER w AS "bundesland_vor"
  FROM bereinigt.abschnitt WHERE lieferung = '2026-02' AND regelsatz = 'v1'
  WINDOW w AS (PARTITION BY vvh_id ORDER BY beginn, ende NULLS LAST)
),
w AS (
  SELECT 'oenace' AS merkmal, "oenace_vor" AS von, "oenace" AS nach, beginn AS tag, vvh_id, bundesland, oenace FROM s WHERE "oenace_vor" <> "oenace" AND beginn BETWEEN '2026-02-01'::date AND '2026-02-28'::date
  UNION ALL
  SELECT 'bundesland' AS merkmal, "bundesland_vor" AS von, "bundesland" AS nach, beginn AS tag, vvh_id, bundesland, oenace FROM s WHERE "bundesland_vor" <> "bundesland" AND beginn BETWEEN '2026-02-01'::date AND '2026-02-28'::date
),
g AS (
  SELECT merkmal, von, nach, tag, count(*)::bigint AS n,
         mode() WITHIN GROUP (ORDER BY bundesland) AS bundesland,
         mode() WITHIN GROUP (ORDER BY oenace) AS oenace
  FROM w GROUP BY merkmal, von, nach, tag HAVING count(*) > 100
)
SELECT g.merkmal, g.von, g.nach, g.tag::text AS tag, g.n, g.bundesland, g.oenace,
       (SELECT jsonb_agg(v ORDER BY v) FROM (
          SELECT w.vvh_id AS v FROM w
          WHERE w.merkmal = g.merkmal AND w.von = g.von AND w.nach = g.nach AND w.tag = g.tag
          ORDER BY 1 LIMIT 5) b) AS beispiele
FROM g ORDER BY g.n DESC, g.merkmal, g.von, g.nach, g.tag
```
